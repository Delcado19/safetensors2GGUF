cmake_minimum_required(VERSION 3.14...3.28)
project(converter_kernel C CXX)

if(NOT DEFINED REFERENCE_SOURCE)
    message(FATAL_ERROR "REFERENCE_SOURCE must point to verified frozen b3962 sources")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option NATIVE AVX AVX2 AVX512 FMA F16C OPENMP CUDA METAL BLAS VULKAN KOMPUTE CCACHE)
    set(GGML_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${REFERENCE_SOURCE}/ggml" ggml)
add_library(converter-kernel SHARED kernel.c)
target_include_directories(converter-kernel PRIVATE "${REFERENCE_SOURCE}/ggml/src")
target_link_libraries(converter-kernel PRIVATE ggml)
if(NOT MSVC)
    target_link_libraries(converter-kernel PRIVATE m)
endif()
